cmake_minimum_required(VERSION 3.20)
project(Tappyweb VERSION 0.2.0 LANGUAGES CXX)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/src/assets.hpp"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/embed.py"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    DEPENDS tools/embed.py assets/htmx.min.js assets/default.css)
add_executable(tappy src/main.cpp src/assets.hpp)
target_compile_features(tappy PRIVATE cxx_std_20)
target_include_directories(tappy PRIVATE vendor)
target_link_libraries(tappy PRIVATE Threads::Threads)
if(WIN32)
    target_compile_definitions(tappy PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(tappy PRIVATE ws2_32)
endif()
if(MINGW)
    target_link_options(tappy PRIVATE -municode -static)
endif()
if(MSVC)
    target_compile_options(tappy PRIVATE /O2 /utf-8 /W4)
    set_property(TARGET tappy PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
    target_compile_options(tappy PRIVATE -O2 -Wall -Wextra)
endif()
install(TARGETS tappy RUNTIME DESTINATION .)
if(WIN32)
    add_executable(tappy-env tools/windows_env.cpp)
    target_compile_features(tappy-env PRIVATE cxx_std_17)
    target_compile_definitions(tappy-env PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(tappy-env PRIVATE advapi32 user32)
    if(MSVC)
        target_compile_options(tappy-env PRIVATE /O2 /utf-8)
        set_property(TARGET tappy-env PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    elseif(MINGW)
        target_link_options(tappy-env PRIVATE -municode -static)
    endif()
endif()
